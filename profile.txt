Set-PSReadLineKeyHandler -Key "RightArrow" -ScriptBlock {
       param($key, $arg)

       $line = $null
       $cursor = $null
       [Microsoft.PowerShell.PSConsoleReadLine]::GetBufferState([ref]$line, [ref]$cursor)

       if ($cursor -lt $line.Length) {
           [Microsoft.PowerShell.PSConsoleReadLine]::ForwardChar($key, $arg)
       } else {
           [Microsoft.PowerShell.PSConsoleReadLine]::AcceptNextSuggestionWord($key, $arg)
       }
}

Set-PSReadLineKeyHandler -Key End -ScriptBlock {
       param($key, $arg)

       $line = $null
       $cursor = $null
       [Microsoft.PowerShell.PSConsoleReadLine]::GetBufferState([ref]$line, [ref]$cursor)

       if ($cursor -lt $line.Length) {
           [Microsoft.PowerShell.PSConsoleReadLine]::EndOfLine($key, $arg)
       } else {
           [Microsoft.PowerShell.PSConsoleReadLine]::AcceptSuggestion($key, $arg)
       }
}

# Run one upgrade step, classify the exit code, return a status record.
# Output is captured (not streamed) so winget skips its console spinner; it is
# echoed only when the step genuinely fails, keeping Success / Nothing-to-upgrade quiet.
function Invoke-UpgradeStep {
    param([string]$Label, [scriptblock]$Action)
    Write-Host "→ $Label" -ForegroundColor Cyan
    $out  = & $Action 2>&1
    $code = $LASTEXITCODE
    # winget's APPINSTALLER_CLI_ERROR_UPDATE_NOT_APPLICABLE = 0x8A15002B
    if     ($code -eq 0)            { $status = 'Success'             }
    elseif ($code -eq -1978335189)  { $status = 'Nothing to upgrade'  }
    else                            { $status = "Failed (exit $code)" }
    if ($status -like 'Failed*') { $out | Out-Host }
    [pscustomobject]@{ Label = $Label; Status = $status }
}

# Like Linux `codeup` but without VS Code extension updates or Claude.
# (yamir/yain are Arch-specific; add winget/scoop steps here if you use them.)
function Invoke-CodeUp {
    $results = @(
        Invoke-UpgradeStep 'Claude Code (winget)' { winget upgrade Anthropic.ClaudeCode }
        Invoke-UpgradeStep 'uv self update'       { uv self update }
        Invoke-UpgradeStep 'bun upgrade'          { bun upgrade }
        Invoke-UpgradeStep 'corepack (npm -g)'    { npm install -g corepack }
        Invoke-UpgradeStep 'lazygit (winget)'     { winget upgrade lazygit }
    )

    Write-Host ''
    Write-Host '── codeup summary ──' -ForegroundColor DarkGray
    foreach ($r in $results) {
        if     ($r.Status -eq 'Success')            { $color = 'Green';  $sym = '✓' }
        elseif ($r.Status -eq 'Nothing to upgrade') { $color = 'Yellow'; $sym = '⚠' }
        else                                        { $color = 'Red';    $sym = '✗' }
        Write-Host ('  {0} {1,-22} {2}' -f $sym, $r.Label, $r.Status) -ForegroundColor $color
    }
}

function Kill-Port {
    param(
        [Parameter(Mandatory=$true)]
        [int]$Port
    )
    $pid = (Get-NetTCPConnection -LocalPort $Port -ErrorAction Stop).OwningProcess[0]
    taskkill /F /PID $pid
}

function notebook2py {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory, Position=0)]
        [string]$NotebookPath
    )

    if (-not (Test-Path -LiteralPath $NotebookPath)) {
        Write-Error "Not found: $NotebookPath"; return
    }
    $nb = Get-Item -LiteralPath $NotebookPath
    if ($nb.Extension -ne '.ipynb') {
        Write-Error "Not an .ipynb file: $($nb.Name)"; return
    }
    $py = Join-Path $nb.DirectoryName "$($nb.BaseName).py"

    # --- ensure required dev tools are in the uv venv ---
    $required = @('jupytext', 'ruff')
    $missing = @()
    foreach ($pkg in $required) {
        & uv pip show $pkg 2>&1 | Out-Null
        if ($LASTEXITCODE -ne 0) { $missing += $pkg }
    }
    if ($missing.Count -gt 0) {
        Write-Host "→ adding missing dev deps: $($missing -join ', ')" -ForegroundColor Yellow
        uv add --dev @missing
        if ($LASTEXITCODE -ne 0) { Write-Error "uv add failed"; return }
    }

    # --- helper: run a step and bail on non-zero exit ---
    function Invoke-Step {
        param([string]$Label, [scriptblock]$Action)
        Write-Host "→ $Label" -ForegroundColor Cyan
        & $Action
        if ($LASTEXITCODE -ne 0) {
            Write-Error "$Label failed (exit $LASTEXITCODE)"
            throw
        }
    }

    try {
        Invoke-Step "converting $($nb.Name)" {
            uv run jupytext --to py:percent $nb.FullName -o $py
        }
        Invoke-Step "stripping notebook artifacts" {
            $stripper = @'
import ast, pathlib, re, sys
p = pathlib.Path(sys.argv[1])
src = p.read_text(encoding="utf-8")
src = re.sub(r"\A# ---\n(?:#[^\n]*\n)*?# ---\n+", "", src)
src = re.sub(r"(?m)^# %%[^\n]*\n", "", src)
tree = ast.parse(src)
remove = set()
for node in tree.body:
    if isinstance(node, ast.Expr) and isinstance(node.value, ast.Name):
        for ln in range(node.lineno, node.end_lineno + 1):
            remove.add(ln)
out = [ln for i, ln in enumerate(src.splitlines(keepends=True), 1) if i not in remove]
p.write_text("".join(out), encoding="utf-8")
'@
            uv run python -c $stripper $py
        }
        Invoke-Step "unused imports + isort (F, I)" {
            uv run ruff check --select F,I --fix $py
        }
        Invoke-Step "formatting" {
            uv run ruff format $py
        }
        Write-Host "✓ $py" -ForegroundColor Green
    } catch {
        # Invoke-Step already reported; exit silently
    }
}
